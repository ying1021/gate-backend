```python
"""认证工具：密码哈希、JWT、输入校验与鉴权规则。"""
import os
import re
import secrets
from datetime import datetime, timedelta, timezone

import bcrypt
import jwt

JWT_SECRET = os.environ.get("JWT_SECRET") or secrets.token_hex(32)
JWT_ALGORITHM = "HS256"
JWT_EXPIRE_DAYS = 7

if not os.environ.get("JWT_SECRET"):
    print("[警告] 未设置 JWT_SECRET，使用临时密钥，服务重启后所有登录会失效")

USERNAME_RE = re.compile(r"^[A-Za-z0-9_]{3,20}$")

# 无需登录即可访问
PUBLIC_PATHS = frozenset({
    "/",
    "/docs",
    "/docs/oauth2-redirect",
    "/openapi.json",
    "/redoc",
    "/auth/register",
    "/auth/login",
})

# 需要登录，但不强制携带 player_id
NO_PLAYER_ID_PATHS = frozenset({
    "/auth/me",
    "/player/list",
    "/player/new_life",
    "/player/create",
    "/game/tick",
    "/game/kill",
    "/game/battle",
    "/game/kill_loot",
    "/shop/info",
})


def validate_credentials(username: str, password: str):
    """返回错误信息；合法时返回 None。"""
    if not USERNAME_RE.match(username or ""):
        return "用户名需为 3-20 位字母、数字或下划线"
    if not password:
        return "密码不能为空"
    length = len(password.encode("utf-8"))
    if length < 6 or length > 72:
        return "密码长度需为 6-72 字节"
    return None


def hash_password(password: str) -> str:
    return bcrypt.hashpw(password.encode("utf-8"), bcrypt.gensalt()).decode("utf-8")


def verify_password(password: str, password_hash: str) -> bool:
    try:
        return bcrypt.checkpw(password.encode("utf-8"), password_hash.encode("utf-8"))
    except (ValueError, TypeError):
        return False


def create_token(user_id: str) -> str:
    now = datetime.now(timezone.utc)
    payload = {"sub": user_id, "iat": now, "exp": now + timedelta(days=JWT_EXPIRE_DAYS)}
    return jwt.encode(payload, JWT_SECRET, algorithm=JWT_ALGORITHM)


def decode_token(token: str) -> dict:
    return jwt.decode(token, JWT_SECRET, algorithms=[JWT_ALGORITHM])


def authenticate_request(path, method, authorization, player_id, owner_lookup):
    """返回 (user_id, error)。

    error 为 None 表示放行；否则为 (status_code, detail)。
    公开路径返回 (None, None)。
    """
    if method == "OPTIONS" or path in PUBLIC_PATHS:
        return None, None

    if not authorization or not authorization.startswith("Bearer "):
        return None, (401, "未登录")

    try:
        payload = decode_token(authorization[7:].strip())
    except Exception:
        return None, (401, "登录已过期，请重新登录")

    user_id = payload.get("sub")
    if not user_id:
        return None, (401, "登录已过期，请重新登录")

    if path not in NO_PLAYER_ID_PATHS and not player_id:
        return None, (400, "缺少 player_id")

    if player_id and owner_lookup(player_id) != user_id:
        return None, (403, "无权访问该角色")

    return user_id, None
```

```python
def create_user(user_id, username, password_hash):
    """创建账号；用户名重复时返回 False"""
    conn = get_connection()
    cursor = conn.cursor()
    try:
        cursor.execute(
            "INSERT INTO users (id, username, password_hash) VALUES (%s, %s, %s)",
            (user_id, username, password_hash),
        )
        conn.commit()
        return True
    except psycopg2.errors.UniqueViolation:
        conn.rollback()
        return False
    finally:
        conn.close()


def get_user_by_username(username):
    conn = get_connection()
    cursor = conn.cursor(cursor_factory=psycopg2.extras.RealDictCursor)
    cursor.execute(
        "SELECT id, username, password_hash FROM users WHERE username = %s",
        (username,),
    )
    row = cursor.fetchone()
    conn.close()
    return dict(row) if row else None


def get_user_by_id(user_id):
    conn = get_connection()
    cursor = conn.cursor(cursor_factory=psycopg2.extras.RealDictCursor)
    cursor.execute("SELECT id, username FROM users WHERE id = %s", (user_id,))
    row = cursor.fetchone()
    conn.close()
    return dict(row) if row else None


def get_player_owner(player_id):
    """返回角色所属用户 ID；角色不存在时返回 None"""
    conn = get_connection()
    cursor = conn.cursor()
    cursor.execute("SELECT user_id FROM players WHERE id = %s", (player_id,))
    row = cursor.fetchone()
    conn.close()
    return row[0] if row else None


def list_players_by_user(user_id):
    conn = get_connection()
    cursor = conn.cursor(cursor_factory=psycopg2.extras.RealDictCursor)
    cursor.execute(
        """
        SELECT id, species, level, current_map, created_at, updated_at
        FROM players
        WHERE user_id = %s
        ORDER BY updated_at DESC
        """,
        (user_id,),
    )
    rows = cursor.fetchall()
    conn.close()
    return [
        {
            "id": row["id"],
            "species": row["species"],
            "level": row["level"],
            "current_map": row["current_map"],
            "created_at": row["created_at"].isoformat() if row["created_at"] else None,
            "updated_at": row["updated_at"].isoformat() if row["updated_at"] else None,
        }
        for row in rows
    ]
```

```python
        method=request.method,
        authorization=request.headers.get("Authorization"),
```

```js
      const map = player.current_map || '未知';
      const time = (player.updated_at || '').replace('T', ' ').slice(0, 16);
      card.textContent = `${player.species}  Lv.${player.level}  · ${map}  · ${time}`;
      card.addEventListener('click', () => this.enterGame(player.id));
      listEl.appendChild(card);
    }
  },

  enterGame: async function(playerId) {
    showCharError('');
    const result = await API.playerState(playerId);
    if (!result || !result['玩家状态']) { showCharError('进入游戏失败，请重试'); return; }

    playerData = result['玩家状态'];
    if (!playerData.inventory) playerData.inventory = {};
    if (!playerData.vital) {
      playerData.vital = { '生命机能': 100, '体力耐力': 100, '饱腹值': 100, '精神状态': 100 };
    }
    if (!playerData.current_map) playerData.current_map = '城郊青草地';

    document.getElementById('charOverlay').classList.add('hidden');
    document.getElementById('authOverlay').classList.add('hidden');
    currentView = 'main';
    clearScreen();
    addMessage('你回到了众生之域。', 'system');
    startTickTimer();
    drawScreen();
  },

  newCharacter: async function() {
    const btn = document.getElementById('newCharBtn');
    showCharError('');
    btn.disabled = true;
    const result = await API.newLife();
    btn.disabled = false;

    if (!result || !result['玩家状态']) { showCharError('创建角色失败，请重试'); return; }

    playerData = result['玩家状态'];
    if (!playerData.inventory) {
      playerData.inventory = { '野果': 2, '草药绷带': 1, '生锈的铁刀': 1 };
    }
    if (!playerData.current_map) playerData.current_map = '城郊青草地';

    document.getElementById('charOverlay').classList.add('hidden');
    document.getElementById('authOverlay').classList.add('hidden');
    currentView = 'main';
    clearScreen();
    addMessage(result['叙事文字'] || result.msg, 'system');
    startTickTimer();
    drawScreen();
  },

  logout: function() {
    clearToken();
    if (typeof stopTickTimer === 'function') stopTickTimer();
    if (typeof clearLocal === 'function') clearLocal();
    playerData = null;
    currentView = 'login';
    this.showLogin();
  },

  setMode(mode) {
    this.mode = mode;
    document.getElementById('tabLogin').classList.toggle('active', mode === 'login');
    document.getElementById('tabRegister').classList.toggle('active', mode === 'register');
    document.getElementById('authPassword2').classList.toggle('hidden', mode !== 'register');
    document.getElementById('authSubmit').textContent = mode === 'register' ? '注册' : '登录';
    document.getElementById('authPassword').autocomplete = mode === 'register' ? 'new-password' : 'current-password';
    showAuthError('');
  },

  async submit() {
    const username = document.getElementById('authUsername').value.trim();
    const password = document.getElementById('authPassword').value;
    showAuthError('');

    if (this.mode === 'register') {
      const password2 = document.getElementById('authPassword2').value;
      if (password !== password2) { showAuthError('两次输入的密码不一致'); return; }
      const result = await API.register(username, password);
      if (result && result.access_token) {
        setToken(result.access_token);
        document.getElementById('authForm').reset();
        await this.showCharacters();
      } else {
        showAuthError((result && result.error) || '注册失败');
      }
      return;
    }

    const result = await API.login(username, password);
    if (result && result.access_token) {
      setToken(result.access_token);
      document.getElementById('authForm').reset();
      await this.showCharacters();
    } else {
      showAuthError((result && result.error) || '登录失败');
    }
  },
};

document.getElementById('tabLogin').addEventListener('click', () => Auth.setMode('login'));
document.getElementById('tabRegister').addEventListener('click', () => Auth.setMode('register'));
document.getElementById('authForm').addEventListener('submit', (event) => {
  event.preventDefault();
  Auth.submit();
});
document.getElementById('newCharBtn').addEventListener('click', () => Auth.newCharacter());
document.getElementById('logoutBtn').addEventListener('click', () => Auth.logout());
```
